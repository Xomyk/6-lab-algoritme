## 📚 Теоретическая часть
*   **Рекурсия** — метод программирования, при котором функция вызывает саму себя. Обязательно должен быть **базовый случай** (условие остановки), иначе произойдет переполнение стека.
*   **Алгоритм Евклида** — эффективный метод вычисления НОД, основанный на том, что `НОД(a, b) = НОД(b, a % b)`.
*   **Ханойские башни** — классическая задача на рекурсию. Для перемещения `n` дисков нужно переместить `n-1` дисков на вспомогательный стержень, затем переложить нижний диск, а потом вернуть `n-1` дисков на целевой.
*   **Сортировки**:
    *   **Пузырьковая:** многократно меняет местами соседние элементы.
    *   **Вставками:** берет элемент и вставляет его в нужное место в уже отсортированной части.

## 💻 Листинг программы (main.cpp)

```cpp
#include <iostream>
#include <vector>

using namespace std;

// ==========================================
// ЧАСТЬ 1: РЕКУРСИВНЫЕ АЛГОРИТМЫ
// ==========================================

// 1. Факториал числа
long long factorial(int n) {
    if (n <= 1) return 1; // Базовый случай
    return n * factorial(n - 1); // Шаг рекурсии
}

// 2. Числа Фибоначчи
int fibonacci(int n) {
    if (n <= 1) return n; // Базовый случай
    return fibonacci(n - 1) + fibonacci(n - 2); // Шаг рекурсии
}

// 3. НОД (Алгоритм Евклида)
int gcd(int a, int b) {
    if (b == 0) return a; // Базовый случай
    return gcd(b, a % b); // Шаг рекурсии
}

// 4. Ханойские башни (для 3 дисков)
void hanoi(int n, char from, char to, char aux) {
    if (n == 1) {
        cout << "Переместить диск 1 с " << from << " на " << to << endl;
        return;
    }
    hanoi(n - 1, from, aux, to); // Шаг 1
    cout << "Переместить диск " << n << " с " << from << " на " << to << endl; // Шаг 2
    hanoi(n - 1, aux, to, from); // Шаг 3
}

// ==========================================
// ЧАСТЬ 2: ПРОСТЫЕ СОРТИРОВКИ С ПОДСЧЕТОМ
// ==========================================
struct Stats {
    int comparisons;
    int swaps;
};

// 1. Пузырьковая сортировка
Stats bubbleSort(int arr[], int n) {
    Stats stats = {0, 0};
    for (int i = 0; i < n - 1; i++) {
        bool swapped = false;
        for (int j = 0; j < n - i - 1; j++) {
            stats.comparisons++;
            if (arr[j] > arr[j + 1]) {
                swap(arr[j], arr[j + 1]);
                stats.swaps++;
                swapped = true;
            }
        }
        if (!swapped) break; // Оптимизация
    }
    return stats;
}

// 2. Сортировка вставками
Stats insertionSort(int arr[], int n) {
    Stats stats = {0, 0};
    for (int i = 1; i < n; i++) {
        int key = arr[i];
        int j = i - 1;
        
        while (j >= 0) {
            stats.comparisons++;
            if (arr[j] > key) {
                arr[j + 1] = arr[j];
                stats.swaps++;
                j--;
            } else {
                break;
            }
        }
        arr[j + 1] = key;
        if (j + 1 != i) {
            stats.swaps++;
        }
    }
    return stats;
}

void printArray(int arr[], int n) {
    for (int i = 0; i < n; i++) cout << arr[i] << " ";
    cout << endl;
}

// ==========================================
// ГЛАВНАЯ ФУНКЦИЯ
// ==========================================
int main() {
    setlocale(LC_ALL, "ru");

    // --- Демонстрация рекурсивных алгоритмов ---
    cout << "=== ЧАСТЬ 1: РЕКУРСИЯ ===" << endl;
    
    int num = 5;
    cout << "1. Факториал " << num << "! = " << factorial(num) << endl;
    
    int fibN = 7;
    cout << "2. Число Фибоначчи (позиция " << fibN << ") = " << fibonacci(fibN) << endl;
    
    int a = 48, b = 18;
    cout << "3. НОД(" << a << ", " << b << ") = " << gcd(a, b) << endl;
    
    cout << "4. Ханойские башни (3 диска):" << endl;
    hanoi(3, 'A', 'C', 'B'); // A - откуда, C - куда, B - вспомогательный
    cout << endl;

    // --- Демонстрация сортировок ---
    cout << "=== ЧАСТЬ 2: СОРТИРОВКИ ===" << endl;
    
    const int N = 8;
    int original[N] = {64, 34, 25, 12, 22, 11, 90, 5};
    
    // Копии для сортировок
    int arrBubble[N], arrInsert[N];
    for (int i = 0; i < N; i++) {
        arrBubble[i] = original[i];
        arrInsert[i] = original[i];
    }

    cout << "Исходный массив: ";
    printArray(original, N);
    cout << endl;

    // Пузырьковая
    Stats statsBubble = bubbleSort(arrBubble, N);
    cout << "Пузырьковая сортировка:" << endl;
    cout << "  Отсортировано: "; printArray(arrBubble, N);
    cout << "  Сравнений: " << statsBubble.comparisons << ", Обменов: " << statsBubble.swaps << endl << endl;

    // Вставками
    Stats statsInsert = insertionSort(arrInsert, N);
    cout << "Сортировка вставками:" << endl;
    cout << "  Отсортировано: "; printArray(arrInsert, N);
    cout << "  Сравнений: " << statsInsert.comparisons << ", Обменов (сдвигов): " << statsInsert.swaps << endl << endl;

    return 0;
